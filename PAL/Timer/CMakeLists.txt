add_library(pal_timer_contract INTERFACE)
target_include_directories(pal_timer_contract INTERFACE include port)

add_library(pal_timer STATIC src/pal_timer.c)
target_link_libraries(pal_timer PUBLIC pal_timer_contract pal_timer_port)

if(CMAKE_CROSSCOMPILING)
    set(timer_port_default none)
else()
    set(timer_port_default host)
endif()
set(TOOLLIB_TIMER_PORT "${timer_port_default}" CACHE STRING "Timer PAL port (host, stm32f4 or none)")
set_property(CACHE TOOLLIB_TIMER_PORT PROPERTY STRINGS host stm32f4 none)

if(TOOLLIB_TIMER_PORT STREQUAL "host")
    add_subdirectory(port/host)
    if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
        add_subdirectory(tests)
    endif()
elseif(TOOLLIB_TIMER_PORT STREQUAL "stm32f4")
    add_subdirectory(port/stm32f4)
elseif(NOT TOOLLIB_TIMER_PORT STREQUAL "none" AND NOT TARGET pal_timer_port)
    message(FATAL_ERROR "Unknown TOOLLIB_TIMER_PORT: ${TOOLLIB_TIMER_PORT}")
endif()
