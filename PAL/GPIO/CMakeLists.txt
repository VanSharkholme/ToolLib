add_library(pal_gpio_contract INTERFACE)
target_include_directories(pal_gpio_contract INTERFACE include port)

add_library(pal_gpio INTERFACE)
target_link_libraries(pal_gpio INTERFACE pal_gpio_contract pal_gpio_port)

if(CMAKE_CROSSCOMPILING)
    set(gpio_port_default none)
else()
    set(gpio_port_default host)
endif()
set(TOOLLIB_GPIO_PORT "${gpio_port_default}" CACHE STRING "GPIO PAL port (host, stm32f4 or none)")
set_property(CACHE TOOLLIB_GPIO_PORT PROPERTY STRINGS host stm32f4 none)

if(TOOLLIB_GPIO_PORT STREQUAL "host")
    add_subdirectory(port/host)
    if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
        add_subdirectory(tests)
    endif()
elseif(TOOLLIB_GPIO_PORT STREQUAL "stm32f4")
    add_subdirectory(port/stm32f4)
elseif(NOT TOOLLIB_GPIO_PORT STREQUAL "none" AND NOT TARGET pal_gpio_port)
    message(FATAL_ERROR "Unknown TOOLLIB_GPIO_PORT: ${TOOLLIB_GPIO_PORT}")
endif()
